# Runtime dependencies — installed into the Astro Runtime image and used by
# Airflow tasks. Keep this file minimal: only what the DAG actually imports
# at execution time. Dev-only tooling (mypy, ruff, test fixtures) lives in
# requirements-dev.txt instead, so it never bloats the production image.

# --- Airflow provider ---
# S3Hook, used by include/chem/s3_io.py to read/write CSVs from S3 or MinIO
# (same code path for both — only the Airflow Connection differs).
apache-airflow-providers-amazon

# --- Cheminformatics / data science stack ---
# RDKit: molecule generation (molzip), property/descriptor calculation.
rdkit
# scikit-learn: K-means clustering + StandardScaler for molecules_clustering.py.
scikit-learn
# pandas: DataFrame handling in graph_generation.py (TMAP/Faerun input format).
pandas
# numpy: feature matrix construction in molecules_clustering.py.
numpy
